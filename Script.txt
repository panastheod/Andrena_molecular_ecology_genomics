with open("test_hwe.hwe", "r") as f:
    # add chr and pos to results for entry whose obs het > 60
    results = []
    # iterate over each line except first header line
    for line in f.readlines()[1:]:
        # extract chr, pos and obs entry

        # extract obs het from obs
        x1, x2, x3 = list(map(int, obs.strip().split("/")))

#!/usr/bin/env python
# coding: utf-8

import os, glob, argparse, csv, re

def read_header_index(filename):
    with open(filename, "r", newline="") as f:
        reader = csv.reader(f, delimiter="\t")
        header = next(reader)
    return {name: i for i, name in enumerate(header)}

def get_ID(filename):
    """
    Build IDs as CHROM:POS:REF:ALT using header-based indexing.
    Works whether POS is present or not; if POS missing, falls back to VCF ID.
    """
    idx = read_header_index(filename)
    ids = ["Pop"]
    with open(filename, "r", newline="") as f:
        reader = csv.reader(f, delimiter="\t")
        _ = next(reader)  # skip header
        for row in reader:
            if "POS" in idx:  # preferred path when POS is present
                chrom = row[idx["#CHROM"]]
                pos   = row[idx["POS"]]
                ref   = row[idx["REF"]]
                alt   = row[idx["ALT"]]
                ids.append(f"{chrom}:{pos}:{ref}:{alt}")
            else:
                # Fallback if POS not in .afreq: use ID plus alleles if available
                if "ID" in idx and row[idx["ID"]] != ".":
                    ids.append(row[idx["ID"]])
                else:
                    chrom = row[idx["#CHROM"]]
                    ref   = row[idx["REF"]]
                    alt   = row[idx["ALT"]]
                    ids.append(f"{chrom}:{ref}:{alt}")
    return ids

def pop_name_from_filename(filename, upper=False, drop_trailing_ED=False):
    base = os.path.basename(filename)
    stem = base[:-len(".afreq")] if base.lower().endswith(".afreq") else os.path.splitext(base)[0]
    if upper: stem = stem.upper()
    if drop_trailing_ED: stem = re.sub(r'ED$', '', stem, flags=re.IGNORECASE)
    return stem

def get_ALT_FREQS(filename, *, upper=False, drop_trailing_ED=False):
    pop = pop_name_from_filename(filename, upper=upper, drop_trailing_ED=drop_trailing_ED)
    idx = read_header_index(filename)
    alt_freq_col = idx.get("ALT_FREQS")
    if alt_freq_col is None:
        raise ValueError(f"ALT_FREQS column not found in {filename}")
    values = [pop]
    with open(filename, "r", newline="") as f:
        reader = csv.reader(f, delimiter="\t")
        _ = next(reader)  # skip header
        for row in reader:
            values.append(row[alt_freq_col])
    return values

if __name__ == "__main__":
    parser = argparse.ArgumentParser(description="Merge .afreq files in a folder.")
    parser.add_argument("dir_path", help="Path to the directory containing .afreq files.")
    parser.add_argument("out_path", help="Output path for merged table.")
    parser.add_argument("--upper", action="store_true", help="Uppercase population names.")
    parser.add_argument("--drop-ed", action="store_true", help="Drop trailing 'ED' from pop names.")
    args = parser.parse_args()

    filenames = sorted(glob.glob(os.path.join(os.path.normpath(args.dir_path), "*.afreq")))
    if not filenames:
        raise SystemExit("No .afreq files found.")

    ID_list = get_ID(filenames[0])
    ALT_FREQS_list = [get_ALT_FREQS(fn, upper=args.upper, drop_trailing_ED=args.drop_ed) for fn in filenames]
    ALT_FREQS_list.sort(key=lambda x: x[0].upper())

    with open(args.out_path, "w", newline="") as out:
        out.write("\t".join(ID_list) + "\n")
        for row in ALT_FREQS_list:
            out.write("\t".join(row) + "\n")

# Fix invalid SNP names (":" etc.)
setnames(gen, make.names(colnames(gen)))

# --- 2. Set parameters ---
chunk_size <- 500
ntree <- 500
corr_thres <- 0.5
maxLevel <- log2(0.368 * nrow(env) / 2)
nchunks <- ceiling(ncol(gen) / chunk_size)

# --- 3. Loop through chunks ---
imp.list <- list()

for (i in seq_len(nchunks)) {
  cat("\n=== Processing chunk", i, "of", nchunks, "===\n")
  
  idx <- ((i - 1) * chunk_size + 1):min(i * chunk_size, ncol(gen))
  gen.sub <- gen[, ..idx]
  data.gf <- cbind(env, gen.sub)
  
  gf <- gradientForest(
    data.gf,
    predictor.vars = colnames(env),
    response.vars  = colnames(gen.sub),
    ntree = ntree,
    maxLevel = maxLevel,
    corr.threshold = corr_thres,
    trace = FALSE
  )
  
  # Extract importance matrix from this chunk
  imp_matrix <- gf$imp.rsq
  imp_dt <- as.data.table(t(imp_matrix))  # transpose: one row per SNP
  setnames(imp_dt, "V1", colnames(env))   # rename column to environmental variable name
  imp_dt[, SNP := rownames(imp_matrix)]
  imp.list[[i]] <- imp_dt
}

# --- 4. Combine all importance tables ---
imp_total <- rbindlist(imp.list, fill = TRUE)

# If you have one environmental variable:
env_name <- colnames(env)[1]
importance_scores <- imp_total[[env_name]]
names(importance_scores) <- imp_total$SNP

# --- 5. Sort and extract top 1% ---
sorted_scores <- sort(importance_scores, decreasing = TRUE)
top_1_percent <- sorted_scores[1:round(length(sorted_scores) * 0.01)]

# --- 6. Save results ---
write.table(
  top_1_percent,
  file = "top_snps_to_extract.txt",
  quote = FALSE,
  row.names = TRUE,
  col.names = FALSE
)

# --- 7. (Optional) Save full importance ranking ---
write.table(
  sorted_scores,
  file = "all_snp_importance_ranked.txt",
  quote = FALSE,
  row.names = TRUE,
  col.names = FALSE
)

cat("\n✅ GradientForest analysis complete. Top 1% SNPs saved to 'top_snps_to_extract.txt'.\n")
